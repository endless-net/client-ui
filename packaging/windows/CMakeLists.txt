cmake_minimum_required(VERSION 3.15)
project(EndlessNetInstaller LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(endlessnet_installer_actions SHARED installer_actions.cpp)
target_compile_features(endlessnet_installer_actions PRIVATE cxx_std_17)
target_compile_definitions(endlessnet_installer_actions PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(endlessnet_installer_actions PRIVATE msi ole32 advapi32 shell32)
